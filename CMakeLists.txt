cmake_minimum_required(VERSION 3.21)
project(OkageNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(okage_assets src/assets.cpp src/model.cpp src/pose.cpp src/animation.cpp src/texture_animation.cpp src/render.cpp src/map.cpp src/event.cpp src/audio.cpp src/sequence.cpp src/bank.cpp src/stream_bank.cpp src/voice.cpp src/music.cpp src/text.cpp src/font.cpp src/dialogue.cpp src/camera.cpp src/field.cpp src/actor.cpp)
target_include_directories(okage_assets PUBLIC include)
target_sources(okage_assets PRIVATE src/effect.cpp)
target_sources(okage_assets PRIVATE src/collision.cpp)
target_sources(okage_assets PRIVATE src/camera_collision.cpp)
target_sources(okage_assets PRIVATE src/map_objects.cpp)
target_sources(okage_assets PRIVATE src/enemy.cpp src/screen_filter.cpp)
target_compile_options(okage_assets PRIVATE -Wall -Wextra -Wpedantic)
add_executable(okage-assets src/assets_main.cpp)
target_link_libraries(okage-assets PRIVATE okage_assets)

if(APPLE)
    enable_language(OBJCXX)
    add_executable(okage-native MACOSX_BUNDLE src/native_runtime.mm src/scene_renderer.mm src/audio_output.cpp)
    target_compile_options(okage-native PRIVATE -fobjc-arc -Wall -Wextra)
    target_link_libraries(okage-native PRIVATE okage_assets "-framework AppKit" "-framework Metal" "-framework AudioToolbox")
