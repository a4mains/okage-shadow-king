cmake_minimum_required(VERSION 3.19)
project(OkageNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(okage_assets src/assets.cpp src/model.cpp src/pose.cpp src/animation.cpp src/texture_animation.cpp src/render.cpp src/map.cpp src/event.cpp src/audio.cpp src/text.cpp)
target_include_directories(okage_assets PUBLIC include)
target_compile_options(okage_assets PRIVATE -Wall -Wextra -Wpedantic)
add_executable(okage-assets src/assets_main.cpp)
target_link_libraries(okage-assets PRIVATE okage_assets)

if(APPLE)
    enable_language(OBJCXX)
    add_executable(okage-render-preview src/render_preview.mm)
    target_compile_options(okage-render-preview PRIVATE -fobjc-arc -Wall -Wextra)
    target_link_libraries(okage-render-preview PRIVATE okage_assets "-framework AppKit" "-framework Metal")
    add_executable(okage-viewer MACOSX_BUNDLE src/viewer.mm)
    target_compile_options(okage-viewer PRIVATE -fobjc-arc)
    target_link_libraries(okage-viewer PRIVATE okage_assets "-framework AppKit")
    set_target_properties(okage-viewer PROPERTIES
        MACOSX_BUNDLE_BUNDLE_NAME "Okage Asset Viewer"
        MACOSX_BUNDLE_GUI_IDENTIFIER "local.okage.asset-viewer")
    add_executable(okage-model-viewer MACOSX_BUNDLE src/model_viewer.mm)
    target_compile_options(okage-model-viewer PRIVATE -fobjc-arc)
    target_link_libraries(okage-model-viewer PRIVATE okage_assets "-framework AppKit")
    set_target_properties(okage-model-viewer PROPERTIES
        MACOSX_BUNDLE_BUNDLE_NAME "Okage Model Research"
        MACOSX_BUNDLE_GUI_IDENTIFIER "local.okage.model-viewer")
endif()

    add_test(NAME event-tests COMMAND event-tests)
    add_executable(map-tests tests/map_tests.cpp)

    add_test(NAME map-tests COMMAND map-tests)
    add_executable(asset-tests tests/asset_tests.cpp)

    add_test(NAME asset-tests COMMAND asset-tests)
    add_executable(model-tests tests/model_tests.cpp)

    add_test(NAME model-tests COMMAND model-tests)
