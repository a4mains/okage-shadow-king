if(NOT TARGET PlayCore OR NOT DEBUGGER_INCLUDED)
    message(FATAL_ERROR "The research oracle requires PlayCore with DEBUGGER_INCLUDED=ON")
endif()
add_executable(okage-pose-oracle "${CMAKE_CURRENT_LIST_DIR}/main.cpp")
set_target_properties(okage-pose-oracle PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/okage-oracle")
target_link_libraries(okage-pose-oracle PRIVATE PlayCore)
target_compile_options(okage-pose-oracle PRIVATE -Wall -Wextra)
target_sources(okage-pose-oracle PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../src/model.cpp"
    "${CMAKE_CURRENT_LIST_DIR}/../../src/assets.cpp" "${CMAKE_CURRENT_LIST_DIR}/../../src/texture_animation.cpp")
target_include_directories(okage-pose-oracle PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../include")
set_target_properties(okage-pose-oracle PROPERTIES CXX_STANDARD 20)
